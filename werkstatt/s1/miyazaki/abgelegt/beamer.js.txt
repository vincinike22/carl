  // Zahlen aus einem Baustein lesen, so tolerant wie möglich: Tage mit Werten oder bloße Tage
  function zahlenAus(bs) {
    var d = (bs && bs.daten) || {}, reihe = [], einheit = d.einheit || '';
    var quelle = Array.isArray(d.eintraege) ? d.eintraege : Array.isArray(d.tage) ? d.tage : Array.isArray(d.perlen) ? d.perlen : Array.isArray(d.werte) ? d.werte : Array.isArray(d) ? d : null;

    quelle.forEach(function (e) {
      if (e == null) return;
      if (typeof e === 'string') { reihe.push({ tag: e.slice(0, 10), v: 1 }); return; }
      if (typeof e === 'number') { reihe.push({ tag: '', v: e }); return; }
      var tag = e.tag || (e.ts ? new Date(e.ts).toISOString().slice(0, 10) : '') || (e.t && /^\d{4}-\d\d-\d\d/.test(e.t) ? e.t.slice(0, 10) : '');
      var v = [e.v, e.wert, e.std, e.stunden, e.h, e.n, e.menge].filter(function (x) { return typeof x === 'number' && isFinite(x); })[0];
      if (v == null) v = 1;
      reihe.push({ tag: tag, v: v });
    });
    if (!reihe.length) return null;
    // je Tag zusammenfassen, letzte 14
    var je = {}, ord = [];
    reihe.forEach(function (r) { var k = r.tag || ('#' + ord.length); if (!(k in je)) { je[k] = 0; ord.push(k); } je[k] += r.v; });
    ord = ord.slice(-14);
    return { punkte: ord.map(function (k) { return { tag: k, v: je[k] }; }), einheit: einheit, name: bs.name || (bs.art || 'Baustein') };
  }

  // Bild verkleinern (längste Seite ≤ 1200 px), als data-URL; bleibt im Gerät
  function verkleinere(f) {
    return new Promise(function (ok) {
      var url = URL.createObjectURL(f), img = new Image();
      img.onload = function () {
        try {
          var m = 1200, s = Math.min(1, m / Math.max(img.naturalWidth, img.naturalHeight));
          var c = D.createElement('canvas'); c.width = Math.max(1, Math.round(img.naturalWidth * s)); c.height = Math.max(1, Math.round(img.naturalHeight * s));
          c.getContext('2d').drawImage(img, 0, 0, c.width, c.height);
          var out = c.toDataURL('image/jpeg', 0.82);
          if (out.length > 700000) { s = 0.6; var c2 = D.createElement('canvas'); c2.width = Math.round(c.width * s); c2.height = Math.round(c.height * s); c2.getContext('2d').drawImage(c, 0, 0, c2.width, c2.height); out = c2.toDataURL('image/jpeg', 0.72); }
          URL.revokeObjectURL(url); ok(out.length > 900000 ? '' : out);
        } catch (e) { URL.revokeObjectURL(url); ok(''); }
      };
      img.onerror = function () { URL.revokeObjectURL(url); ok(''); };
      img.src = url;
    });
  }
